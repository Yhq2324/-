**C++实现文件：**

```cpp
// VehiclePartInfoWidget.cpp
#include "VehiclePartInfoWidget.h"
#include "Components/TextBlock.h"
#include "Components/Image.h"
#include "Components/ScrollBox.h"
#include "Components/Button.h"

void UVehiclePartInfoWidget::NativeConstruct()
{
    Super::NativeConstruct();
    
    // 绑定按钮事件
    if (CloseButton)
    {
        CloseButton->OnClicked.AddDynamic(this, &UVehiclePartInfoWidget::OnCloseButtonClicked);
    }
    
    bIsVisible = false;
    SetVisibility(ESlateVisibility::Hidden);
}

void UVehiclePartInfoWidget::ShowPartInfo(const FPartInfo& PartInfo)
{
    CurrentPartInfo = PartInfo;
    
    // 更新UI内容
    UpdateTextContent();
    UpdateIcon();
    PopulateSpecsList();
    
    // 播放显示动画
    SetVisibility(ESlateVisibility::Visible);
    PlayShowAnimation();
    bIsVisible = true;
}

void UVehiclePartInfoWidget::HidePartInfo()
{
    PlayHideAnimation();
    
    // 动画结束后隐藏
    FTimerHandle TimerHandle;
    GetWorld()->GetTimerManager().SetTimer(TimerHandle, [this]()
    {
        SetVisibility(ESlateVisibility::Hidden);
        bIsVisible = false;
    }, 0.3f, false);
}

void UVehiclePartInfoWidget::UpdateTextContent()
{
    if (PartNameText)
    {
        PartNameText->SetText(FText::FromString(CurrentPartInfo.PartName));
    }
    
    if (DescriptionText)
    {
        DescriptionText->SetText(CurrentPartInfo.Description);
    }
}

void UVehiclePartInfoWidget::UpdateIcon()
{
    if (PartIcon && CurrentPartInfo.Icon)
    {
        PartIcon->SetBrushFromTexture(CurrentPartInfo.Icon);
    }
}

void UVehiclePartInfoWidget::PopulateSpecsList()
{
    if (!SpecsScrollBox) return;
    
    // 清空现有内容
    SpecsScrollBox->ClearChildren();
    
    // 动态创建规格文本
    for (const FString& Spec : CurrentPartInfo.TechnicalSpecs)
    {
        UTextBlock* SpecText = NewObject<UTextBlock>(this);
        SpecText->SetText(FText::FromString(Spec));
        SpecsScrollBox->AddChild(SpecText);
    }
}

void UVehiclePartInfoWidget::OnCloseButtonClicked()
{
    HidePartInfo();
}

void UVehiclePartInfoWidget::PlayShowAnimation()
{
    if (ShowAnimation)
    {
        PlayAnimation(ShowAnimation);
    }
}

void UVehiclePartInfoWidget::PlayHideAnimation()
{
    if (HideAnimation)
    {
        PlayAnimation(HideAnimation);
    }
}
```
