```cpp
// AssetLoader.cpp
void UAssetLoader::LoadMeshAsync(TSoftObjectPtr<UStaticMesh> MeshRef, 
                                  TFunction<void(UStaticMesh*)> OnLoaded)
{
    if (MeshRef.IsNull())
    {
        OnLoaded(nullptr);
        return;
    }

    // 如果已加载，直接返回
    if (MeshRef.IsValid())
    {
        OnLoaded(MeshRef.Get());
        return;
    }

    // 异步加载
    FStreamableManager& Streamable = UAssetManager::GetStreamableManager();
    TSharedPtr<FStreamableHandle> Handle = Streamable.RequestAsyncLoad(
        MeshRef.ToSoftObjectPath(),
        [this, MeshRef, OnLoaded]()
        {
            UStaticMesh* LoadedMesh = MeshRef.Get();
            OnLoaded(LoadedMesh);
        }
    );

    ActiveHandles.Add(Handle);
}
```

```cpp
// CameraController.cpp
void ACameraController::FocusOnPart(APartActor* Part)
{
    if (!Part) return;

    // 计算最佳观察位置
    FVector PartLocation = Part->GetActorLocation();
    FBox PartBounds = Part->GetComponentsBoundingBox();
    float PartSize = PartBounds.GetSize().Size();

    // 目标位置：零件前方一定距离
    TargetLocation = PartLocation - GetActorForwardVector() * (PartSize * 2.0f);
    TargetRotation = (PartLocation - TargetLocation).Rotation();

    bIsFocusing = true;
    FocusAnimTime = 0.0f;
}

void ACameraController::TickFocusAnimation(float DeltaTime)
{
    if (!bIsFocusing) return;

    FocusAnimTime += DeltaTime;
    float Alpha = FMath::Clamp(FocusAnimTime / FocusAnimDuration, 0.0f, 1.0f);

    // Ease In-Out插值
    Alpha = FMath::InterpEaseInOut(0.0f, 1.0f, Alpha, 2.0f);

    // 平滑移动
    FVector NewLocation = FMath::Lerp(GetActorLocation(), TargetLocation, Alpha);
    FRotator NewRotation = FMath::Lerp(GetActorRotation(), TargetRotation, Alpha);

    SetActorLocationAndRotation(NewLocation, NewRotation);

    if (Alpha >= 1.0f)
    {
        bIsFocusing = false;
    }
}
```

```cpp
// InteractionSystem.cpp
void UInteractionSystem::HandleMouseClick()
{
    FVector HitLocation;
    APartActor* ClickedPart = RaycastPart(HitLocation);

    if (ClickedPart)
    {
        // 取消之前的选中
        if (SelectedPart && SelectedPart != ClickedPart)
        {
            SelectedPart->SetState(EPartState::Idle);
        }

        // 选中新零件
        SelectedPart = ClickedPart;
        SelectedPart->SetState(EPartState::Selected);

        // 触发事件
        OnPartSelected.Broadcast(SelectedPart);
    }
}

APartActor* UInteractionSystem::RaycastPart(FVector& HitLocation)
{
    APlayerController* PC = GetWorld()->GetFirstPlayerController();
    if (!PC) return nullptr;

    FVector2D MousePosition;
    PC->GetMousePosition(MousePosition.X, MousePosition.Y);

    FVector WorldLocation, WorldDirection;
    PC->DeprojectScreenPositionToWorld(
        MousePosition.X, MousePosition.Y,
        WorldLocation, WorldDirection
    );

    FHitResult HitResult;
    FCollisionQueryParams Params;
    Params.AddIgnoredActor(PC->GetPawn());

    bool bHit = GetWorld()->LineTraceSingleByChannel(
        HitResult,
        WorldLocation,
        WorldLocation + WorldDirection * 10000.0f,
        ECC_Visibility,
        Params
    );

    if (bHit)
    {
        HitLocation = HitResult.Location;
        return Cast<APartActor>(HitResult.GetActor());
    }

    return nullptr;
}
```

## ✅ 总结

### **积木式开发的优势：**

1. ✅ **独立开发** - 每个模块可以单独编写和测试
2. ✅ **并行工作** - 多人可以同时开发不同模块
3. ✅ **易于调试** - 问题隔离在单个模块内
4. ✅ **可复用** - 模块可以用在其他项目
5. ✅ **灵活组合** - 可选模块可以轻松添加或移除
